# js事件监听逻辑
先名爵我们需要读取的输入分别是
  1. ```<input id="save-realm" type="checkbox">```
  2. ```<input id="film-mode" type="checkbox">```
需要注意的是在film-mode切换时,切换```<label class="switch-wrapper" for="film-mode">```的```innerHTML```为```拍照|录像```

在此基础上设计报文格式为
```http
http:\\${basehost}\film?filmmode=${el.id}&saverealm=${el.id}
```
对应的, 摄像头的工作模式选择为mode: cap(capture)|rcd(record)
        保存位置选择为saverealm: server(SD.h)|client|both
所以我们所发送的合法指令一共有6条
```http
http:\\192.108.1.104\film?filmmode=cap&saverealm=server
http:\\192.108.1.104\film?filmmode=cap&saverealm=client
http:\\192.108.1.104\film?filmmode=cap&saverealm=both

http:\\192.108.1.104\film?filmmode=rcd&saverealm=server
http:\\192.108.1.104\film?filmmode=rcd&saverealm=client
http:\\192.108.1.104\film?filmmode=rcd&saverealm=both
```

// 内容放在document.addEventListener('DOMContentLoaded', function (event) {}里面
// 设置监听端口为
//   var baseHost = document.location.origin
//   var streamUrl = baseHost + ':81'
const save_realm = document.getElementById('save-realm')
const film_mode = document.getElementById('film-mode')

function send_filmmode_and_saverealm () {
  let realm = save_realm.value 
  let mode = film_mode.value
  const query = `${baseHost}/film?mode=${mode}&saverealm=${realm}`
  fetch(query)
  .then()
}

save_realm.onchange = () => {
  send_filmmode_and_saverealm ()
}
film_mode.onchange = () => {
  send_filmmode_and_saverealm ()
  if(film-mode.value == "cap"){film_mode.innerHTML = '拍照'}
  // 注意这里是伪代码, checkbox的value只有0和1
  elseif(film-mode.value == "rcd"){film_mode.innerHTML = '录像'}
}
