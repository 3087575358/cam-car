# API 手册
*这里只写关于录像的前后端逻辑, 代码整合留给后面*
    ## 功能：film（拍照/录像）

    ### 端点
    GET /film

    ### 参数
    | 参数名 | 必填 | 值范围 | 语义 |
    |--------|------|--------|------|
    | filmmode | 是 | cap / rcd | 功能模式 |
    | saverealm | 是 | client / server / both | 保存位置 |
    | streaming | 否 | on / off | 录像意图（rcd模式有效）|

    ### 状态流转
    | 状态 | 含义 | 服务器标识 |
    |------|------|-----------|
    | IDLE | 未录制，可接受开始命令 | `g_rec_state == 0` |
    | RECORDING | 正在录制，可接受停止命令 | `g_rec_state == 1` |

    ### 响应
    /*
    当前状态 + 命令 → 下一状态 + 响应
    IDLE    + streaming=on  → RECORDING + OK:started:{filename}
    IDLE    + streaming=off → IDLE      + ERR:not_recording
    RECORDING + streaming=on  → RECORDING + ERR:already_recording
    RECORDING + streaming=off → IDLE    + OK:stopped:{filename}
    */
    OK:started:{filename}
    OK:stopped:{filename}
    ERR:already_recording
    ERR:not_recording
    /*
    idle-activity 窝工活动/停工行为
    */

# film_handler
```cpp
// ========== 对照模板：状态定义 ==========
enum StreamState { IDLE, RECORDING };
StreamState g_stream_state = IDLE;
char g_filename[32] = {0};

esp_err_t handle_film(httpd_req_t* req) {
    char buf_mode[8] = {0}, buf_realm[16] = {0}, buf_stream[8] = {0};
    
    // ========== 对照模板：解析参数 ==========
    httpd_query_key_value(req->uri, "filmmode",  buf_mode,  sizeof(buf_mode));
    httpd_query_key_value(req->uri, "saverealm", buf_realm, sizeof(buf_realm));
    
    // streaming 不管rcd模式是否开启,都进行streaming数据获取
    int has_streaming = (httpd_query_key_value(req->uri, "streaming", buf_stream, sizeof(buf_stream)) == ESP_OK);
    
    // ========== 对照模板：状态流转逻辑 ==========
    // 正常逻辑状态IDLE --streaming=on--> RECORDING --streaming=off--> IDLE
    
    if (strcmp(buf_mode, "rcd") == 0) {
        bool rcd_switch;
        
        if (has_streaming) {
            rcd_switch = (strcmp(buf_stream, "on") == 0);
        } else {
            // 未读到streaming数据, 直接关闭录像, 并且默认不允许打开录像
            rcd_switch = (g_stream_state == IDLE);
        }
        
        char response[64];
        
        if (rcd_switch) {
            // IDLE --streaming=on--> RECORDING
            if (g_stream_state == RECORDING) {
                // 模板响应：ERR:already_recording
                httpd_resp_send(req, "ERR:already_recording", 23);
                return ESP_OK;
            }
            g_stream_state = RECORDING;
            sprintf(g_filename, "/sd/VID_%03d.avi", get_next_id());
            start_avi(g_filename);
            \\ start_avi暂时还没写, 伪代码, 等待日后补齐
            
            // 模板响应：OK:started:{filename}
            sprintf(response, "OK:started:%s", g_filename);
            
        } else {
            // RECORDING --streaming=off--> IDLE
            if (g_stream_state == IDLE) {
                // 模板响应：ERR:not_recording
                httpd_resp_send(req, "ERR:not_recording", 19);
                return ESP_OK;
            }
            stop_avi();
            g_stream_state = IDLE;
            
            // 模板响应：OK:stopped:{filename}
            sprintf(response, "OK:stopped:%s", g_filename);
        }
        
        httpd_resp_send(req, response, strlen(response));
        return ESP_OK;
    }
    
    httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "Unknown mode");
    return ESP_FAIL;
}
```

```js
        }
```
