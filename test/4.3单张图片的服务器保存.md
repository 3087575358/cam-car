# 以客户端本地存放照片为例
```http
http:\\192.108.1.104\film?filmmode=cap&saverealm=server
```

我们先注册film域
```cpp
void startCameraServer()
{
    http_config_t config = HTTPD_DEFAULT_CONFIG();//80,32768
    config.max_uri_handlers = 16;

    httpd_uri_t film_uri={
        .uri= "/film",
        .method = HTTP_GET,
        .handler = film_handler,
        .user_ctx = NULL
#ifdef CONFIG_HTTPD_WS_SUPPORT
        ,
        .is_websocket = true,
        .handle_ws_control_frames = false,
        .supported_subprotocol = NULL
#endif
    }

    if (httpd_start(&stream_httpd, &config) == ESP_OK)
    {
        http_register_uri_handler(stream_httpd, &stream_uri);
    }
}
```

然后进行命令解析
```cpp
static esp_err_t film_handler(httpd_req_t *req){
    char *req_buf = NULL;
    char film_mode[32];
    char save_realm[32];

    camera_fb_t *fb = NULL;
    esp_err_t res = ESP_OK;

    if (parse_get(req, &req_buf) != ESP_OK) {
    return ESP_FAIL;
    }

    if (httpd_query_key_value(buf, "filmmode", film_mode, sizeof(film_mode)) != ESP_OK ||
    httpd_query_key_value(buf, "saverealm", save_realm, sizeof(save_realm)) != ESP_OK) {
        free(buf);
        httpd_resp_send_404(req);
        return ESP_FAIL;

    if (!strcmp(film_mode, "cap") && !strcmp(save_realm, "server")) {
        // 拍照(保持相机画面)
        fb = esp_camera_fb_get();
        // 存入SD卡
        save_photo_to_server(fb);
        // 返回相机画面
        esp_camera_fb_return(fb);
        // 清理内存并返回ESP_OK
        free(buf);
        return ESP_OK;
    }


}
```

将照片存入SD卡
```cpp
#ifdef SD_MMC_HOST_TYPE_SD
#include "FS.h"                // SD Card ESP32
#include "SD_MMC.h"            // SD Card ESP32
#endif

bool sd_ready = false;

void setup_sd_card() {
  // 如果担心新版报错，用条件编译
#if ESP_ARDUINO_VERSION_MAJOR < 2
    sd_ready = SD_MMC.begin(true);
#else
    sd_ready = SD_MMC.begin("/sdcard", true);
#endif
}

void save_photo_to_server(camera_fb_t *fb) {
  if (!sd_ready || !fb) return;
    
    String filename = "/IMG_" + String(millis()) + ".jpg";//文件类型+文件时间+文件格式
    
    File file = SD_MMC.open(filename.c_str(), FILE_WRITE);
    if (!file) return;
    
    size_t written = file.write(fb->buf, fb->len);
    file.close();
    
    if (written != fb->len) SD_MMC.remove(filename.c_str());  // 写失败删残片
}
```
