  <script>
    document.addEventListener('DOMContentLoaded', function (event) {
      // 设置监听端口为
      //   var baseHost = document.location.origin
      //   var streamUrl = baseHost + ':80'
      const save_realm = document.getElementById('save-realm')
      const film_mode = document.getElementById('film-mode')
      const filmModeText = document.getElementById('film-mode-text')

      // 加入初始化文本，与 checkbox 状态同步
      filmModeText.textContent = film_mode.checked ? '拍照' : '录像'

      function send_filmmode_and_saverealm() {
        let realm = save_realm.value
        let mode = film_mode.checked ? "cap" : "rcd"
        filmModeText.textContent = mode === "cap" ? '拍照' : '录像'
        
        const query = `${baseHost}/film?mode=${mode}&saverealm=${realm}`
        fetch(query)
          .catch(e => console.error(e))
      }

      save_realm.onchange = send_filmmode_and_saverealm
      film_mode.onchange = send_filmmode_and_saverealm
    })
  </script>
